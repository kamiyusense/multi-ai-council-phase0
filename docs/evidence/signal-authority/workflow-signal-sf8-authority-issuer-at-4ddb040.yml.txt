# Owner-authority issuer. A manual (workflow_dispatch) run on main by the one
# allowlisted human yields sanitized GitHub OIDC identity evidence. It writes
# nothing, uses no secrets, and issues no permission other than "this human
# dispatched this exact main commit". It does not touch Issue #33 or runtime state.
name: Signal SF-8 authority issuer
on:
  workflow_dispatch:
permissions:
  contents: read
  id-token: write
concurrency:
  group: signal-sf8-authority-issuer
  cancel-in-progress: false
jobs:
  issue:
    runs-on: ubuntu-latest
    timeout-minutes: 5
    if: github.repository == 'kamiyusense/multi-ai-council-phase0' && github.ref == 'refs/heads/main' && github.run_attempt == 1 && github.event_name == 'workflow_dispatch' && contains(fromJSON('["20355232"]'), github.actor_id)
    steps:
      - name: Checkout the dispatched main commit without persisted credentials
        uses: actions/checkout@11d5960a326750d5838078e36cf38b85af677262
        with:
          ref: ${{ github.sha }}
          persist-credentials: false
      - name: Request once, verify GitHub signature and context in memory
        env:
          PYTHONDONTWRITEBYTECODE: '1'
          OIDC_PROOF_PROFILE: sf8-authority
        run: node automation/command_bus/github_oidc_signature_proof.mjs
      - name: Save sanitized evidence only
        if: always()
        uses: actions/upload-artifact@ea165f8d65b6e75b540449e92b4886f43607fa02
        with:
          name: sf8-authority-evidence-${{ github.run_id }}
          path: ${{ runner.temp }}/oidc-proof-evidence.json
          if-no-files-found: error
          retention-days: 30
